import AppKit
import Combine
import CoreMedia
import Darwin
import ScreenCaptureKit

enum ShotterScrollCaptureError: LocalizedError {
  case invalidArea
  case areaTooLarge
  case displayUnavailable
  case noFrames

  var errorDescription: String? {
    switch self {
    case .invalidArea:
      String(localized: "Choose an area of at least 80 × 80 points within one display.")
    case .areaTooLarge:
      String(localized: "Choose a smaller area for a scrolling screenshot.")
    case .displayUnavailable:
      String(localized: "The display changed. Start a new scrolling screenshot.")
    case .noFrames:
      String(localized: "No frames were captured. Try selecting the area again.")
    }
  }
}

/// Captures a fixed viewport while the user scrolls the underlying application.
/// The border never accepts input, and the controls do not activate FancyTool.
@MainActor
final class ShotterScrollCaptureController {
  var onComplete: ((CGImage, CGFloat) -> Void)?
  var onCancel: (() -> Void)?
  var onError: ((Error) -> Void)?

  private enum State { case idle, starting, recording, interrupted, finishing, stopped }

  private var state = State.idle
  private var stream: SCStream?
  private var frameProcessor: ShotterScrollFrameProcessor?
  private var controls: ShotterScrollControlsPanel?
  private var border: NSPanel?
  private var screenObserver: AnyCancellable?
  private var snapshotRecoveryTask: Task<Void, Never>?
  private var streamWatchdogTask: Task<Void, Never>?
  private var scrollMonitor: Any?
  private var lastScrollUptime: TimeInterval = 0
  private var lastStreamRestartUptime: TimeInterval = 0
  private var isRestartingStream = false
  private var displayID: CGDirectDisplayID?
  private var displayFrame = NSRect.zero
  private var displayScale: CGFloat = 1
  private var outputScale: CGFloat = 1
  private var capturedHeight = 0
  private var startupFailure: Error?
  private var captureFilter: SCContentFilter?
  private var captureConfiguration: SCStreamConfiguration?
  private var snapshotConfiguration: SCStreamConfiguration?
  private var latestProgress: ShotterScrollFrameProcessor.Progress?

  func start(in rect: NSRect, targetApp: NSRunningApplication?) async throws {
    guard state == .idle else { throw CancellationError() }
    guard rect.width.isFinite, rect.height.isFinite,
          rect.width >= 80, rect.height >= 80,
          let screen = NSScreen.screens.first(where: { $0.frame.contains(rect) }),
          let identifier = ShotterCoordinateSpace.displayID(for: screen),
          let sourceRect = ShotterCoordinateSpace.displayLocalScreenCaptureRect(fromAppKit: rect, on: screen) else {
      throw ShotterScrollCaptureError.invalidArea
    }

    state = .starting
    displayID = identifier
    displayFrame = screen.frame
    displayScale = screen.backingScaleFactor
    showWindows(around: rect, on: screen)
    scrollMonitor = NSEvent.addGlobalMonitorForEvents(matching: .scrollWheel) { [weak self] _ in
      guard let self, rect.contains(NSEvent.mouseLocation) else { return }
      self.lastScrollUptime = ProcessInfo.processInfo.systemUptime
    }
    observeDisplayChanges()

    do {
      // Exclude only this session's UI. The selected content may belong to
      // FancyTool itself, so excluding the entire application loses that view.
      let content = try await SCShareableContent.excludingDesktopWindows(false, onScreenWindowsOnly: true)
      try checkStarting()
      guard let display = content.displays.first(where: { $0.displayID == identifier }) else {
        throw ShotterScrollCaptureError.displayUnavailable
      }
      let sessionWindowIDs = Set([border, controls].compactMap { window in
        window.map { CGWindowID($0.windowNumber) }
      })
      let sessionWindows = content.windows.filter { sessionWindowIDs.contains($0.windowID) }
      let filter = SCContentFilter(display: display, excludingWindows: sessionWindows)
      outputScale = max(1, CGFloat(filter.pointPixelScale))
      let width = max(1, Int((sourceRect.width * outputScale).rounded(.up)))
      let height = max(1, Int((sourceRect.height * outputScale).rounded(.up)))
      guard width <= 40_000_000 / height else { throw ShotterScrollCaptureError.areaTooLarge }

      let configuration = SCStreamConfiguration()
      configuration.sourceRect = sourceRect
      configuration.width = width
      configuration.height = height
      configuration.scalesToFit = false
      configuration.showsCursor = false
      configuration.capturesAudio = false
      configuration.pixelFormat = kCVPixelFormatType_32BGRA
      configuration.colorSpaceName = CGColorSpace.sRGB
      configuration.queueDepth = 8
      configuration.minimumFrameInterval = CMTime(value: 1,
        timescale: CMTimeScale(max(60, screen.maximumFramesPerSecond)))
      captureFilter = filter
      captureConfiguration = configuration
      let snapshotConfiguration = SCStreamConfiguration()
      snapshotConfiguration.sourceRect = sourceRect
      snapshotConfiguration.width = width
      snapshotConfiguration.height = height
      snapshotConfiguration.scalesToFit = false
      snapshotConfiguration.showsCursor = false
      snapshotConfiguration.pixelFormat = kCVPixelFormatType_32BGRA
      snapshotConfiguration.colorSpaceName = CGColorSpace.sRGB
      self.snapshotConfiguration = snapshotConfiguration

      let events = ShotterScrollEventRelay { [weak self] event in self?.receive(event) }
      let processor = ShotterScrollFrameProcessor { events.send($0) }
      let stream = SCStream(filter: filter, configuration: configuration, delegate: processor)
      processor.setActiveStream(stream)
      self.frameProcessor = processor
      self.stream = stream
      try stream.addStreamOutput(processor, type: .screen, sampleHandlerQueue: processor.deliveryQueue)
      if let targetApp, targetApp.processIdentifier != ProcessInfo.processInfo.processIdentifier,
         !targetApp.isTerminated {
        targetApp.activate(options: [])
      }
      try await stream.startCapture()
      try checkStarting()
      if let startupFailure { throw startupFailure }
      processor.commitActiveStream(stream)
      state = .recording
      startSnapshotRecovery()
      startStreamWatchdog()
      if let latestProgress {
        showProgress(latestProgress)
      }
    } catch {
      stopResources()
      state = .stopped
      throw error
    }
  }

  func cancel() {
    guard state != .stopped else { return }
    state = .stopped
    let callback = onCancel
    clearCallbacks()
    stopResources()
    callback?()
  }

  private func checkStarting() throws {
    try Task.checkCancellation()
    guard state == .starting else { throw CancellationError() }
  }

  private func receive(_ event: ShotterScrollFrameProcessor.Event) {
    guard state == .starting || state == .recording || state == .finishing || state == .interrupted else { return }
    switch event {
    case let .progress(update, width, height, _, _):
      guard state == .starting || state == .recording else { return }
      let progress = ShotterScrollFrameProcessor.Progress(update: update, width: width, height: height)
      guard progress != latestProgress else { return }
      latestProgress = progress
      // A stream may deliver its first buffer before startCapture returns.
      // Keep the controls in their preparing state until startup completes.
      guard state == .recording else { return }
      showProgress(progress)
    case let .preview(preview):
      controls?.setPreview(preview.image)
    case let .failure(message):
      guard state != .interrupted else { return }
      let error = NSError(domain: "FancyTool.ScrollingCapture", code: 1,
                          userInfo: [NSLocalizedDescriptionKey: message])
      if state == .starting {
        // A delegate failure can race with a successful startCapture return.
        // Do not silently enter a recording state with an already-stopped stream.
        startupFailure = error
      } else if state == .finishing {
        return
      } else {
        interruptOrFail(error)
      }
    }
  }

  private func showProgress(_ progress: ShotterScrollFrameProcessor.Progress) {
    capturedHeight = progress.height
    controls?.setCapturedSize(width: progress.width, height: progress.height)
    controls?.finishButton.isEnabled = progress.height > 0
    let message: String
    switch progress.update {
    case .appended, .unchanged:
      message = String(localized: "Scroll down slowly inside the selected area.")
    case .noOverlap:
      message = String(localized: "Unable to match the latest frame.")
    case .limitReached:
      // Reaching the limit must not silently export a truncated screenshot.
      pauseCapture(message: String(localized: "Size limit reached. Finish to keep the preview, or cancel and select a smaller area."))
      return
    }
    controls?.setStatus(message, height: progress.height)
  }

  private func finish() {
    guard state == .recording || state == .interrupted, let frameProcessor else { return }
    state = .finishing
    snapshotRecoveryTask?.cancel()
    snapshotRecoveryTask = nil
    streamWatchdogTask?.cancel()
    streamWatchdogTask = nil
    screenObserver = nil
    if let scrollMonitor { NSEvent.removeMonitor(scrollMonitor) }
    scrollMonitor = nil
    controls?.finishButton.isEnabled = false
    controls?.setStatus(String(localized: "Finishing…"), height: capturedHeight)
    let stream = self.stream
    self.stream = nil
    Task { try? await stream?.stopCapture() }
    Task { [weak self, frameProcessor] in
      await frameProcessor.finishPendingFrames()
      let image = await frameProcessor.makeImage()
      guard let self, self.state == .finishing else { return }
      guard let image else {
        self.fail(ShotterScrollCaptureError.noFrames)
        return
      }
      let callback = self.onComplete
      self.clearCallbacks()
      self.state = .stopped
      self.frameProcessor = nil
      self.captureFilter = nil
      self.captureConfiguration = nil
      self.snapshotConfiguration = nil
      self.closeWindows()
      callback?(image, self.outputScale)
    }
  }

  private func fail(_ error: Error) {
    guard state != .stopped else { return }
    state = .stopped
    let callback = onError
    clearCallbacks()
    stopResources()
    callback?(error)
  }

  private func interruptOrFail(_ error: Error) {
    guard state == .recording, capturedHeight > 0 else {
      fail(error)
      return
    }
    // A permission/display interruption should not throw away the part that
    // was already collected. The user can still finish or cancel explicitly.
    pauseCapture(message: String(localized: "Capture stopped. Finish to keep the captured content."))
    print("[Shotter] scrolling capture interrupted: \(error.localizedDescription)")
  }

  private func pauseCapture(message: String) {
    state = .interrupted
    snapshotRecoveryTask?.cancel()
    snapshotRecoveryTask = nil
    streamWatchdogTask?.cancel()
    streamWatchdogTask = nil
    if let scrollMonitor { NSEvent.removeMonitor(scrollMonitor) }
    scrollMonitor = nil
    screenObserver = nil
    frameProcessor?.stopAcceptingFrames()
    frameProcessor?.refreshPreview()
    let stream = self.stream
    self.stream = nil
    Task { [stream] in try? await stream?.stopCapture() }
    controls?.setStatus(message, height: capturedHeight)
    controls?.finishButton.isEnabled = capturedHeight > 0
  }

  private func clearCallbacks() {
    onComplete = nil
    onCancel = nil
    onError = nil
  }

  private func stopResources() {
    snapshotRecoveryTask?.cancel()
    snapshotRecoveryTask = nil
    streamWatchdogTask?.cancel()
    streamWatchdogTask = nil
    if let scrollMonitor { NSEvent.removeMonitor(scrollMonitor) }
    scrollMonitor = nil
    screenObserver = nil
    captureFilter = nil
    captureConfiguration = nil
    snapshotConfiguration = nil
    let processor = frameProcessor
    processor?.stopAcceptingFrames()
    frameProcessor = nil
    let stream = self.stream
    self.stream = nil
    closeWindows()
    Task { [stream, processor] in
      try? await stream?.stopCapture()
      processor?.discard()
    }
  }

  private func observeDisplayChanges() {
    screenObserver = NotificationCenter.default.publisher(for: NSApplication.didChangeScreenParametersNotification)
      .receive(on: RunLoop.main)
      .sink { [weak self] _ in
        guard let self, self.state == .starting || self.state == .recording else { return }
        let screen = NSScreen.screens.first { ShotterCoordinateSpace.displayID(for: $0) == self.displayID }
        guard let screen, screen.frame == self.displayFrame,
              screen.backingScaleFactor == self.displayScale else {
          self.interruptOrFail(ShotterScrollCaptureError.displayUnavailable)
          return
        }
      }
  }

  private func startSnapshotRecovery() {
    snapshotRecoveryTask?.cancel()
    snapshotRecoveryTask = Task { [weak self] in
      var failures = 0
      while !Task.isCancelled {
        try? await Task.sleep(for: .milliseconds(100))
        guard let self, self.state == .recording, let frameProcessor = self.frameProcessor,
              let filter = self.captureFilter,
              let configuration = self.snapshotConfiguration else { return }
        guard frameProcessor.needsSnapshotRecovery() else { continue }
        let requestedAt = ProcessInfo.processInfo.systemUptime
        let requestedAtMach = mach_absolute_time()
        do {
          let image = try await SCScreenshotManager.captureImage(
            contentFilter: filter, configuration: configuration)
          guard self.state == .recording else { return }
          failures = 0
          let progress = await frameProcessor.appendRecoveryImage(
            image, requestedAt: requestedAt, captureTime: requestedAtMach)
          if progress?.progress.update == .appended,
            frameProcessor.lastStreamFrameAge() > 0.7 {
            await self.restartStream()
          }
        } catch {
          failures += 1
          print("[Shotter] scrolling snapshot recovery failed: \(error.localizedDescription)")
          if failures >= 3 {
            failures = 0
            await self.restartStream()
          }
        }
      }
    }
  }

  private func startStreamWatchdog() {
    streamWatchdogTask?.cancel()
    streamWatchdogTask = Task { [weak self] in
      while !Task.isCancelled {
        try? await Task.sleep(for: .milliseconds(200))
        guard let self, self.state == .recording, let processor = self.frameProcessor else { return }
        let now = ProcessInfo.processInfo.systemUptime
        guard now - self.lastScrollUptime < 1,
          now - self.lastStreamRestartUptime > 3,
          processor.lastStreamFrameAge() > 0.7
        else { continue }
        await self.restartStream()
      }
    }
  }

  private func restartStream() async {
    guard state == .recording, !isRestartingStream,
      let oldStream = stream, let filter = captureFilter,
      let configuration = captureConfiguration, let processor = frameProcessor,
      ProcessInfo.processInfo.systemUptime - lastStreamRestartUptime > 3 else { return }
    isRestartingStream = true
    lastStreamRestartUptime = ProcessInfo.processInfo.systemUptime
    controls?.setStatus(String(localized: "Capture stalled. Reconnecting…"), height: capturedHeight)
    let replacement = SCStream(filter: filter, configuration: configuration, delegate: processor)
    do {
      try replacement.addStreamOutput(processor, type: .screen,
        sampleHandlerQueue: processor.deliveryQueue)
      processor.setActiveStream(replacement)
      try await replacement.startCapture()
      guard state == .recording else {
        processor.abandonPendingStream(replacement)
        try? await replacement.stopCapture()
        isRestartingStream = false
        return
      }
      stream = replacement
      processor.commitActiveStream(replacement)
      if let latestProgress { showProgress(latestProgress) }
      Task { try? await oldStream.stopCapture() }
    } catch {
      processor.abandonPendingStream(replacement)
      try? await replacement.stopCapture()
      if let latestProgress { showProgress(latestProgress) }
      print("[Shotter] scrolling stream restart failed: \(error.localizedDescription)")
    }
    isRestartingStream = false
  }

  private func showWindows(around rect: NSRect, on screen: NSScreen) {
    let border = NSPanel(contentRect: rect.insetBy(dx: -3, dy: -3),
                         styleMask: [.borderless, .nonactivatingPanel], backing: .buffered, defer: false)
    border.isReleasedWhenClosed = false
    border.isOpaque = false
    border.backgroundColor = .clear
    border.hasShadow = false
    border.ignoresMouseEvents = true
    border.hidesOnDeactivate = false
    border.level = .screenSaver
    border.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .stationary, .ignoresCycle]
    let borderView = NSView(frame: NSRect(origin: .zero, size: border.frame.size))
    borderView.wantsLayer = true
    borderView.layer?.borderColor = NSColor.controlAccentColor.cgColor
    borderView.layer?.borderWidth = 2
    borderView.layer?.cornerRadius = 5
    border.contentView = borderView
    border.orderFrontRegardless()
    self.border = border

    let controls = ShotterScrollControlsPanel(availableSize: NSSize(
      width: screen.visibleFrame.width - 24, height: screen.visibleFrame.height - 24))
    controls.onFinish = { [weak self] in self?.finish() }
    controls.onCancel = { [weak self] in self?.cancel() }
    let area = screen.visibleFrame
    let size = controls.frame.size
    controls.setFrameOrigin(Self.controlsOrigin(size: size, selection: rect, visibleFrame: area))
    controls.orderFrontRegardless()
    self.controls = controls
  }

  static func controlsOrigin(size: NSSize, selection rect: NSRect, visibleFrame area: NSRect) -> NSPoint {
    let minimumX = area.minX + 8
    let maximumX = max(minimumX, area.maxX - size.width - 8)
    let minimumY = area.minY + 8
    let maximumY = max(minimumY, area.maxY - size.height - 8)
    let x = min(max(rect.midX - size.width / 2, minimumX), maximumX)
    let y = min(max(rect.midY - size.height / 2, minimumY), maximumY)
    // Keep the live preview beside the capture. For wide selections, use the
    // space above or below instead; none of these placements intercept scrolls.
    if rect.maxX + 12 <= maximumX {
      return NSPoint(x: rect.maxX + 12, y: y)
    }
    if rect.minX - size.width - 12 >= minimumX {
      return NSPoint(x: rect.minX - size.width - 12, y: y)
    }
    if rect.minY - size.height - 12 >= minimumY {
      return NSPoint(x: x, y: rect.minY - size.height - 12)
    }
    if rect.maxY + 12 <= maximumY {
      return NSPoint(x: x, y: rect.maxY + 12)
    }
    return NSPoint(x: maximumX, y: minimumY)
  }

  private func closeWindows() {
    if let scrollMonitor { NSEvent.removeMonitor(scrollMonitor) }
    scrollMonitor = nil
    let windows = [border, controls].compactMap { $0 }
    border = nil
    controls = nil
    windows.forEach { $0.orderOut(nil) }
    // Keep ARC ownership through AppKit's deferred display/close transaction.
    DispatchQueue.main.async {
      windows.forEach { $0.close() }
      DispatchQueue.main.asyncAfter(deadline: .now() + 1) {
        withExtendedLifetime(windows) {}
      }
    }
  }
}

private final class ShotterScrollEventRelay: @unchecked Sendable {
  private let lock = NSLock()
  private var progress: ShotterScrollFrameProcessor.Event?
  private var preview: ShotterScrollFrameProcessor.Event?
  private var failure: ShotterScrollFrameProcessor.Event?
  private var scheduled = false
  private let receive: @MainActor @Sendable (ShotterScrollFrameProcessor.Event) -> Void

  init(receive: @escaping @MainActor @Sendable (ShotterScrollFrameProcessor.Event) -> Void) {
    self.receive = receive
  }

  func send(_ event: ShotterScrollFrameProcessor.Event) {
    let needsDelivery = lock.withLock {
      switch event {
      case .progress: progress = event
      case .preview: preview = event
      case .failure: failure = event
      }
      guard !scheduled else { return false }
      scheduled = true
      return true
    }
    guard needsDelivery else { return }
    DispatchQueue.main.async { [self] in
      let events = lock.withLock {
        let events = [progress, preview, failure].compactMap { $0 }
        progress = nil
        preview = nil
        failure = nil
        scheduled = false
        return events
      }
      events.forEach { receive($0) }
    }
  }
}

private final class ShotterScrollFrameProcessor: NSObject, SCStreamOutput, SCStreamDelegate, @unchecked Sendable {
  enum Update: Equatable, Sendable { case appended, unchanged, noOverlap, limitReached }
  struct Progress: Equatable, Sendable {
    let update: Update
    let width: Int
    let height: Int
  }
  enum Event: Sendable {
    case progress(Update, Int, Int, UInt64, Bool)
    case preview(Preview)
    case failure(String)
  }

  // CGImage is immutable; only this wrapper crosses back to the main actor.
  struct Preview: @unchecked Sendable { let image: CGImage }
  private struct Sample: @unchecked Sendable {
    let image: CGImage
    let captureTime: UInt64

    init?(buffer: CMSampleBuffer, captureTime: UInt64) {
      guard let pixels = buffer.imageBuffer,
            CVPixelBufferGetPixelFormatType(pixels) == kCVPixelFormatType_32BGRA,
            CVPixelBufferLockBaseAddress(pixels, .readOnly) == kCVReturnSuccess else { return nil }
      defer { CVPixelBufferUnlockBaseAddress(pixels, .readOnly) }
      let width = CVPixelBufferGetWidth(pixels)
      let height = CVPixelBufferGetHeight(pixels)
      let rowBytes = CVPixelBufferGetBytesPerRow(pixels)
      guard width > 0, height > 0, width <= 40_000_000 / height,
            rowBytes >= width * 4, rowBytes <= Int.max / height,
            let baseAddress = CVPixelBufferGetBaseAddress(pixels) else { return nil }
      let data = Data(bytes: baseAddress, count: rowBytes * height)
      guard let provider = CGDataProvider(data: data as CFData),
            let image = CGImage(width: width, height: height,
                                bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: rowBytes,
                                space: CGColorSpace(name: CGColorSpace.sRGB)!,
                                bitmapInfo: CGBitmapInfo.byteOrder32Little.union(CGBitmapInfo(rawValue: CGImageAlphaInfo.premultipliedFirst.rawValue)),
                                provider: provider, decode: nil, shouldInterpolate: false,
                                intent: .defaultIntent) else { return nil }
      self.image = image
      self.captureTime = captureTime
    }
  }

  let deliveryQueue = DispatchQueue(label: "com.wu.FancyTool.scrolling.delivery", qos: .userInitiated)
  private let processingQueue = DispatchQueue(label: "com.wu.FancyTool.scrolling.stitching", qos: .userInitiated)
  private let lock = NSLock()
  private var isAccepting = true
  private var isBusy = false
  private var pendingSamples: [Sample?] = []
  private var pendingSampleIndex = 0
  private var maximumBatchCount = 8
  private var activeStreamID: ObjectIdentifier?
  private var pendingStreamID: ObjectIdentifier?
  private var lastStreamFrameUptime: TimeInterval = 0
  private var lastAcceptedCaptureTime: UInt64 = 0
  private var lastAppendMatched = false
  private var stitcher = ShotterScrollStitcher()
  private var previewScheduled = false
  private var previewDirty = false
  private var lastPreviewUptime: TimeInterval = 0
  private var lastGrowthUptime: TimeInterval = 0
  private var firstUnmatchedUptime: TimeInterval = 0
  private var consecutiveUnmatchedFrames = 0
  private let receive: @Sendable (Event) -> Void

  init(receive: @escaping @Sendable (Event) -> Void) { self.receive = receive }

  func setActiveStream(_ stream: SCStream) {
    lock.withLock {
      pendingStreamID = ObjectIdentifier(stream)
      lastStreamFrameUptime = ProcessInfo.processInfo.systemUptime
    }
  }

  func commitActiveStream(_ stream: SCStream) {
    lock.withLock {
      activeStreamID = ObjectIdentifier(stream)
      pendingStreamID = nil
      lastStreamFrameUptime = ProcessInfo.processInfo.systemUptime
    }
  }

  func abandonPendingStream(_ stream: SCStream) {
    lock.withLock {
      if pendingStreamID == ObjectIdentifier(stream) { pendingStreamID = nil }
    }
  }

  func lastStreamFrameAge() -> TimeInterval {
    lock.withLock { ProcessInfo.processInfo.systemUptime - lastStreamFrameUptime }
  }

  func lastGrowthAge() -> TimeInterval {
    lock.withLock { ProcessInfo.processInfo.systemUptime - lastGrowthUptime }
  }

  func stream(_ stream: SCStream, didOutputSampleBuffer sampleBuffer: CMSampleBuffer, of type: SCStreamOutputType) {
    guard type == .screen, sampleBuffer.isValid,
          let attachments = CMSampleBufferGetSampleAttachmentsArray(sampleBuffer, createIfNecessary: false) as? [[SCStreamFrameInfo: Any]],
          let rawStatus = attachments.first?[.status] as? Int,
          rawStatus == SCFrameStatus.complete.rawValue || rawStatus == SCFrameStatus.started.rawValue ||
            rawStatus == SCFrameStatus.idle.rawValue,
          lock.withLock({ isAccepting &&
            (activeStreamID == ObjectIdentifier(stream) || pendingStreamID == ObjectIdentifier(stream)) })
    else { return }
    lock.withLock {
      if activeStreamID == ObjectIdentifier(stream) {
        lastStreamFrameUptime = ProcessInfo.processInfo.systemUptime
      }
    }
    guard rawStatus != SCFrameStatus.idle.rawValue else { return }
    autoreleasepool {
      let captureTime = (attachments.first?[.displayTime] as? NSNumber)?.uint64Value
        ?? mach_absolute_time()
      guard let sample = Sample(buffer: sampleBuffer, captureTime: captureTime) else { return }
      enqueue(sample)
    }
  }

  private func enqueue(_ sample: Sample) {
    lock.withLock {
      guard isAccepting else { return }
      if isBusy {
        pendingSamples.append(sample)
        return
      }
      isBusy = true
      processingQueue.async { [self, sample] in
        process(sample)
        while let batch = lock.withLock({ () -> [Sample]? in
          guard pendingSampleIndex < pendingSamples.count else {
            pendingSamples.removeAll(keepingCapacity: true)
            pendingSampleIndex = 0
            isBusy = false
            return nil
          }
          let end = min(pendingSamples.count, pendingSampleIndex + maximumBatchCount)
          var batch: [Sample] = []
          for index in pendingSampleIndex..<end {
            guard let sample = pendingSamples[index] else { break }
            if let previous = batch.last, sample.captureTime <= previous.captureTime { break }
            batch.append(sample)
          }
          return batch
        }) {
          var index = batch.count - 1
          var fellBack = false
          let previousHeight = stitcher.pixelHeight
          while !process(batch[index], speculative: index > 0) {
            fellBack = true
            index /= 2
          }
          let consumedCount = index + 1
          let growth = stitcher.pixelHeight - previousHeight
          if growth > 0 {
            let pointsPerFrame = max(1, (growth + consumedCount - 1) / consumedCount)
            let predictedCount = max(1, min(32, batch[index].image.height / 3 / pointsPerFrame))
            maximumBatchCount = fellBack ? min(predictedCount, consumedCount) : predictedCount
          } else {
            maximumBatchCount = 1
          }
          lock.withLock {
            let end = pendingSampleIndex + index + 1
            for consumed in pendingSampleIndex..<end { pendingSamples[consumed] = nil }
            pendingSampleIndex = end
            if pendingSampleIndex >= 64 {
              pendingSamples.removeFirst(pendingSampleIndex)
              pendingSampleIndex = 0
            }
          }
        }
      }
    }
  }

  @discardableResult
  private func process(_ sample: Sample, speculative: Bool = false) -> Bool {
    autoreleasepool {
      guard sample.captureTime > lastAcceptedCaptureTime else { return !speculative }
      let result = stitcher.append(sample.image,
        maximumDisplacement: speculative ? sample.image.height / 3 : nil)
      guard !speculative || result == .appended else { return false }
      let progress = progress(for: result)
      if lastAppendMatched { lastAcceptedCaptureTime = sample.captureTime }
      receive(.progress(progress.update, progress.width, progress.height,
        sample.captureTime, lastAppendMatched))
      if progress.update == .appended || previewDirty { schedulePreview() }
      return true
    }
  }

  private func append(_ image: CGImage, suppressTransientMismatch: Bool = true) -> Progress {
    progress(for: stitcher.append(image), suppressTransientMismatch: suppressTransientMismatch)
  }

  private func progress(for result: ShotterScrollStitcher.Update,
    suppressTransientMismatch: Bool = true) -> Progress {
    var update: Update
    switch result {
    case .appended: update = .appended; lastAppendMatched = true
    case .unchanged: update = .unchanged; lastAppendMatched = true
    case .noOverlap: update = .noOverlap; lastAppendMatched = false
    case .limitReached: update = .limitReached; lastAppendMatched = false
    }
    if update == .appended {
      consecutiveUnmatchedFrames = 0
      lock.withLock { lastGrowthUptime = ProcessInfo.processInfo.systemUptime }
    } else if update == .noOverlap {
      if consecutiveUnmatchedFrames == 0 {
        firstUnmatchedUptime = ProcessInfo.processInfo.systemUptime
      }
      consecutiveUnmatchedFrames += 1
      if suppressTransientMismatch && (consecutiveUnmatchedFrames < 3 ||
        ProcessInfo.processInfo.systemUptime - firstUnmatchedUptime < 0.35) {
        update = .unchanged
      }
    } else if update == .unchanged {
      consecutiveUnmatchedFrames = 0
    }
    return Progress(update: update, width: stitcher.pixelWidth, height: stitcher.pixelHeight)
  }

  func needsSnapshotRecovery() -> Bool {
    lock.withLock { isAccepting &&
      ProcessInfo.processInfo.systemUptime - lastGrowthUptime >= 0.15 &&
      ProcessInfo.processInfo.systemUptime - lastStreamFrameUptime >= 0.45 }
  }

  func appendRecoveryImage(_ image: CGImage, requestedAt: TimeInterval,
    captureTime: UInt64) async -> (progress: Progress, matched: Bool)? {
    let snapshot = Preview(image: image)
    return await withCheckedContinuation { continuation in
      processingQueue.async { [self, snapshot] in
        if lock.withLock({ isAccepting && lastGrowthUptime <= requestedAt }) &&
          captureTime > lastAcceptedCaptureTime {
          let progress = autoreleasepool { append(snapshot.image) }
          if lastAppendMatched { lastAcceptedCaptureTime = captureTime }
          receive(.progress(progress.update, progress.width, progress.height,
            captureTime, lastAppendMatched))
          if progress.update == .appended || previewDirty { schedulePreview() }
          continuation.resume(returning: (progress, lastAppendMatched))
        } else {
          continuation.resume(returning: nil)
        }
      }
    }
  }

  /// Render at most ten small thumbnails per second on the processing queue.
  /// A delayed refresh still runs after the last scroll frame, so the preview
  /// cannot remain one update behind when scrolling stops.
  private func schedulePreview() {
    previewDirty = true
    let now = ProcessInfo.processInfo.systemUptime
    let delay = max(0, 0.1 - (now - lastPreviewUptime))
    if delay == 0 {
      // A busy drain can outlive the timer queued behind it. Publish here so
      // preview updates also arrive during continuous scrolling.
      publishPreview()
    } else if !previewScheduled {
      previewScheduled = true
      processingQueue.asyncAfter(deadline: .now() + delay) { [self] in
        previewScheduled = false
        if previewDirty { schedulePreview() }
      }
    }
  }

  private func publishPreview(force: Bool = false) {
    guard force || lock.withLock({ isAccepting }),
          let image = stitcher.makePreview(maximumSize: CGSize(width: 448, height: 480)) else { return }
    lastPreviewUptime = ProcessInfo.processInfo.systemUptime
    previewDirty = false
    receive(.preview(Preview(image: image)))
  }

  func stream(_ stream: SCStream, didStopWithError error: Error) {
    guard lock.withLock({ activeStreamID == ObjectIdentifier(stream) }) else { return }
    stopAcceptingFrames()
    receive(.failure(error.localizedDescription))
  }

  func stopAcceptingFrames() { lock.withLock { isAccepting = false } }
  func finishPendingFrames() async {
    await withCheckedContinuation { continuation in
      deliveryQueue.async { [self] in
        stopAcceptingFrames()
        processingQueue.async { continuation.resume() }
      }
    }
  }
  func resumeAcceptingFrames() { lock.withLock { isAccepting = true } }
  func refreshPreview() { processingQueue.async { [self] in publishPreview(force: true) } }

  func appendFinalImage(_ image: CGImage) async -> Progress {
    let snapshot = Preview(image: image)
    return await withCheckedContinuation { continuation in
      processingQueue.async { [self, snapshot] in
        let progress = autoreleasepool { append(snapshot.image, suppressTransientMismatch: false) }
        publishPreview(force: true)
        continuation.resume(returning: progress)
      }
    }
  }

  func makeImage() async -> CGImage? {
    await withCheckedContinuation { continuation in
      processingQueue.async { [self] in
        publishPreview(force: true)
        let image = autoreleasepool { stitcher.makeImage() }
        stitcher = ShotterScrollStitcher()
        continuation.resume(returning: image)
      }
    }
  }

  func discard() {
    processingQueue.async { [self] in stitcher = ShotterScrollStitcher() }
  }
}

@MainActor
private final class ShotterScrollSurfaceView: NSView {
  override func hitTest(_ point: NSPoint) -> NSView? { nil }

  override func draw(_ dirtyRect: NSRect) {
    let shape = NSBezierPath(roundedRect: bounds.insetBy(dx: 0.5, dy: 0.5), xRadius: 18, yRadius: 18)
    NSGraphicsContext.saveGraphicsState()
    shape.addClip()
    NSColor.black.withAlphaComponent(0.38).setFill()
    shape.fill()
    NSGradient(colors: [NSColor.systemCyan.withAlphaComponent(0.12),
                        NSColor.systemIndigo.withAlphaComponent(0.04),
                        NSColor.black.withAlphaComponent(0.08)])?.draw(in: bounds, angle: -70)
    NSGraphicsContext.restoreGraphicsState()
    NSColor.white.withAlphaComponent(0.16).setStroke()
    shape.lineWidth = 1
    shape.stroke()
  }
}

@MainActor
private final class ShotterScrollCanvasView: NSView {
  override var mouseDownCanMoveWindow: Bool { true }

  override func draw(_ dirtyRect: NSRect) {
    let shape = NSBezierPath(roundedRect: bounds.insetBy(dx: 0.5, dy: 0.5), xRadius: 12, yRadius: 12)
    NSGradient(starting: NSColor.black.withAlphaComponent(0.30),
               ending: NSColor.black.withAlphaComponent(0.16))?.draw(in: shape, angle: -90)
    NSGraphicsContext.saveGraphicsState()
    shape.addClip()
    NSColor.white.withAlphaComponent(0.035).setFill()
    for x in stride(from: CGFloat(9), to: bounds.width, by: 14) {
      for y in stride(from: CGFloat(9), to: bounds.height, by: 14) {
        NSBezierPath(ovalIn: NSRect(x: x, y: y, width: 1, height: 1)).fill()
      }
    }
    NSGraphicsContext.restoreGraphicsState()
    NSColor.white.withAlphaComponent(0.08).setStroke()
    shape.lineWidth = 1
    shape.stroke()
  }
}

@MainActor
private final class ShotterScrollActionButton: NSButton {
  private let prominent: Bool

  init(title: String, prominent: Bool) {
    self.prominent = prominent
    super.init(frame: .zero)
    self.title = title
    font = .systemFont(ofSize: 12, weight: .semibold)
    isBordered = false
    setButtonType(.momentaryPushIn)
  }

  required init?(coder: NSCoder) { fatalError("init(coder:) has not been implemented") }

  override var isEnabled: Bool {
    didSet { needsDisplay = true }
  }

  override func draw(_ dirtyRect: NSRect) {
    let shape = NSBezierPath(roundedRect: bounds.insetBy(dx: 0.5, dy: 0.5), xRadius: 9, yRadius: 9)
    if prominent && isEnabled {
      let opacity = isHighlighted ? 0.70 : 0.92
      NSGradient(starting: NSColor.controlAccentColor.withAlphaComponent(opacity),
                 ending: NSColor.systemIndigo.withAlphaComponent(opacity))?.draw(in: shape, angle: -25)
    } else {
      NSColor.white.withAlphaComponent(isHighlighted ? 0.14 : (isEnabled ? 0.08 : 0.04)).setFill()
      shape.fill()
    }
    NSColor.white.withAlphaComponent(isEnabled ? 0.16 : 0.06).setStroke()
    shape.lineWidth = 1
    shape.stroke()
    let attributes: [NSAttributedString.Key: Any] = [
      .font: font ?? NSFont.systemFont(ofSize: 12, weight: .semibold),
      .foregroundColor: NSColor.white.withAlphaComponent(isEnabled ? 0.95 : 0.28)
    ]
    let label = title as NSString
    let labelSize = label.size(withAttributes: attributes)
    label.draw(at: NSPoint(x: (bounds.width - labelSize.width) / 2,
                           y: (bounds.height - labelSize.height) / 2), withAttributes: attributes)
  }
}
